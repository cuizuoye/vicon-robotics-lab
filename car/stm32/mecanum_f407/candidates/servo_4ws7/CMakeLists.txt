cmake_minimum_required(VERSION 3.20)
option(SERVO_ARM "Compile independent locked OFFLINE image only" OFF)
option(SERVO_SANITIZERS "Native ASan/UBSan" ON)
get_filename_component(BASE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
get_filename_component(WORKSPACE "${BASE}/../.." ABSOLUTE)
set(ACCEPTED ${BASE}/candidates/continuous_4wgc)
file(READ ${CMAKE_CURRENT_LIST_DIR}/upstream.json manifest)
string(JSON count LENGTH "${manifest}")
math(EXPR last "${count}-1")
foreach(index RANGE ${last})
  string(JSON file MEMBER "${manifest}" ${index})
  string(JSON expected GET "${manifest}" "${file}")
  file(SHA256 ${WORKSPACE}/${file} actual)
  if(NOT actual STREQUAL expected)
    message(FATAL_ERROR "Protected source drift: ${file}; stop without patching baseline")
  endif()
endforeach()
if(SERVO_ARM)
  include(${BASE}/cmake/target.cmake)
  include(${BASE}/cmake/gcc-arm-none-eabi.cmake)
  set(CMAKE_C_COMPILER_FORCED TRUE)
  include(${ACCEPTED}/cmake/absolute_compiler.cmake)
endif()
project(servo_4ws7_OFFLINE C ASM)
set(DEFS FOUR_LINK_MOTION=1 FOUR_LINK_GROUNDB=1 DRIVE_BENCH=1
    DRIVE_STARTUP_OBSERVE=1 FOUR_SERVO_CANDIDATE=1)
set(ALGORITHMS Src/four_wheel_speed.c Src/four_drive_group.c Src/drive_bench_control.c
    Src/four_link_motion.c Src/servo_policy.c ${BASE}/Src/mecanum_drive_plan.c
    ${BASE}/Src/mecanum_kinematics.c ${BASE}/Src/mc_protocol.c)
if(SERVO_ARM)
  set(CMAKE_PROJECT_INCLUDE_BEFORE ${ACCEPTED}/cmake/absolute_compiler.cmake)
  add_subdirectory(${BASE} ${CMAKE_CURRENT_BINARY_DIR}/legacy-metadata EXCLUDE_FROM_ALL)
  unset(CMAKE_PROJECT_INCLUDE_BEFORE)
  get_target_property(legacy_sources mecanum_f407_four_link_groundb SOURCES)
  set(REPLACED Src/four_link_motion.c Src/four_drive_group.c Src/four_wheel_speed.c
      Src/drive_bench_control.c Src/mecanum_drive_plan.c Src/mecanum_kinematics.c
      Src/mc_protocol.c Src/four_link_bench_main.c)
  file(READ ${BASE}/Src/four_link_bench_main.c main_source)
  string(REPLACE "READY,drive_pi_start8_v1" "READY,OFFLINE_4WS7_LOCKED" main_source "${main_source}")
  file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/four_link_bench_main_4ws7.c "${main_source}")
  set(sources ${ALGORITHMS} ${CMAKE_CURRENT_BINARY_DIR}/four_link_bench_main_4ws7.c)
  foreach(file IN LISTS legacy_sources)
    if(file IN_LIST REPLACED)
      continue()
    elseif(IS_ABSOLUTE "${file}")
      list(APPEND sources ${file})
    else()
      list(APPEND sources ${BASE}/${file})
    endif()
  endforeach()
  set(TARGET mecanum_f407_servo_4ws7_LOCKED_OFFLINE)
  add_executable(${TARGET} ${sources})
  foreach(property INCLUDE_DIRECTORIES COMPILE_DEFINITIONS COMPILE_OPTIONS)
    get_target_property(values mecanum_f407_four_link_groundb ${property})
    set_property(TARGET ${TARGET} PROPERTY ${property} "${values}")
  endforeach()
  get_target_property(options mecanum_f407_four_link_groundb LINK_OPTIONS)
  list(FILTER options EXCLUDE REGEX "^-Wl,-Map=")
  target_link_options(${TARGET} PRIVATE ${options} -Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/${TARGET}.map)
  set_target_properties(${TARGET} PROPERTIES SUFFIX ".elf")
  add_custom_command(TARGET ${TARGET} POST_BUILD
      COMMAND ${CMAKE_OBJCOPY} -O binary $<TARGET_FILE:${TARGET}> ${TARGET}.bin
      COMMAND ${CMAKE_SIZE} $<TARGET_FILE:${TARGET}>
      COMMENT "LOCKED OFFLINE 4WS7 ONLY; NO HARDWARE EXECUTION OR FLASH COMMAND")
else()
  enable_testing()
  foreach(TARGET test_servo_4ws7 test_servo_locked)
    if("${TARGET}" STREQUAL "test_servo_4ws7")
      set(test_source tests/test_servo.c)
    else()
      set(test_source tests/test_locked.c)
    endif()
    add_executable(${TARGET} ${test_source} ${ALGORITHMS})
    if("${TARGET}" STREQUAL "test_servo_4ws7")
      target_compile_definitions(${TARGET} PRIVATE SERVO_NATIVE_TEST=1)
    endif()
    target_include_directories(${TARGET} BEFORE PRIVATE ${CMAKE_CURRENT_LIST_DIR}/Inc)
    target_include_directories(${TARGET} PRIVATE ${BASE}/Inc ${WORKSPACE}/protocol)
    target_compile_definitions(${TARGET} PRIVATE ${DEFS})
    target_compile_options(${TARGET} PRIVATE -std=c11 -Wall -Wextra -Wpedantic -Werror -g)
    target_link_libraries(${TARGET} PRIVATE m)
    if(SERVO_SANITIZERS)
      target_compile_options(${TARGET} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
      target_link_options(${TARGET} PRIVATE -fsanitize=address,undefined -no-pie)
    endif()
    add_test(NAME ${TARGET} COMMAND ${TARGET})
  endforeach()
endif()
if(SERVO_ARM)
  target_include_directories(${TARGET} BEFORE PRIVATE ${CMAKE_CURRENT_LIST_DIR}/Inc)
  target_include_directories(${TARGET} PRIVATE ${BASE}/Inc ${WORKSPACE}/protocol)
  target_compile_definitions(${TARGET} PRIVATE ${DEFS})
  target_link_libraries(${TARGET} PRIVATE m)
endif()
