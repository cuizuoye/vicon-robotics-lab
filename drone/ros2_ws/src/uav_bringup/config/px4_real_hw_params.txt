EKF2_EVP_NOISE  0.01        # 位置噪声下限 [m]
EKF2_EVA_NOISE  0.05        # 姿态噪声下限 [rad]
EKF2_EVV_NOISE  0.01        # 速度噪声下限 [m/s]
EKF2_EV_DELAY   <按实测>     # [ms] 动捕->飞控链路延时，0~300，影响跟踪品质
EKF2_EV_POS_X   <按实测>     # 动捕刚体中心相对重心的偏移 [m]（前）
EKF2_EV_POS_Y   <按实测>     # 右
EKF2_EV_POS_Z   <按实测>     # 下
