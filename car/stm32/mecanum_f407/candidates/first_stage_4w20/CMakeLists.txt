cmake_minimum_required(VERSION 3.20)
option(FIRST_STAGE_ARM "Build separate 4W20 artifact, never flash" OFF)
get_filename_component(BASE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
get_filename_component(WORKSPACE "${BASE}/../.." ABSOLUTE)
set(ACCEPTED ${BASE}/candidates/continuous_4wgc)
set(HIGH ${BASE}/candidates/high_speed_4wgd)
foreach(manifest ${ACCEPTED}/baseline.json ${HIGH}/accepted_4wgc.json)
  file(READ ${manifest} baseline)
  string(JSON count LENGTH "${baseline}")
  math(EXPR last "${count}-1")
  if(manifest STREQUAL "${ACCEPTED}/baseline.json")
    set(root ${BASE})
  else()
    set(root ${ACCEPTED})
  endif()
  foreach(index RANGE ${last})
    string(JSON file MEMBER "${baseline}" ${index})
    string(JSON expected GET "${baseline}" "${file}")
    file(SHA256 ${root}/${file} actual)
    if(NOT actual STREQUAL expected)
      message(FATAL_ERROR "Accepted source drift: ${root}/${file}")
    endif()
  endforeach()
endforeach()
if(FIRST_STAGE_ARM)
  include(${BASE}/cmake/target.cmake)
  include(${BASE}/cmake/gcc-arm-none-eabi.cmake)
  set(CMAKE_C_COMPILER_FORCED TRUE)
  include(${ACCEPTED}/cmake/absolute_compiler.cmake)
endif()
project(first_stage_4w20 C ASM)
file(READ ${ACCEPTED}/Src/drive_bench_control.c banner)
string(REPLACE "OFFLINE_drive_continuous_4wgc_candidate" "commissioning_first_stage_4w20" banner "${banner}")
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/drive_bench_control.c "${banner}")
set(ALGORITHMS ${HIGH}/Src/four_wheel_speed.c
    ${CMAKE_CURRENT_LIST_DIR}/Src/mecanum_drive_plan.c
    ${ACCEPTED}/Src/four_drive_group.c ${ACCEPTED}/Src/four_link_motion.c
    ${CMAKE_CURRENT_BINARY_DIR}/drive_bench_control.c
    ${BASE}/Src/mecanum_kinematics.c ${BASE}/Src/mc_protocol.c)
if(FIRST_STAGE_ARM)
  set(CMAKE_PROJECT_INCLUDE_BEFORE ${ACCEPTED}/cmake/absolute_compiler.cmake)
  add_subdirectory(${BASE} ${CMAKE_CURRENT_BINARY_DIR}/legacy-metadata EXCLUDE_FROM_ALL)
  unset(CMAKE_PROJECT_INCLUDE_BEFORE)
  get_target_property(legacy_sources mecanum_f407_four_link_groundb SOURCES)
  set(REPLACED Src/four_link_motion.c Src/four_drive_group.c Src/four_wheel_speed.c
      Src/drive_bench_control.c Src/mecanum_drive_plan.c Src/mecanum_kinematics.c Src/mc_protocol.c)
  set(sources ${ALGORITHMS})
  foreach(file IN LISTS legacy_sources)
    if(file IN_LIST REPLACED)
      continue()
    elseif(IS_ABSOLUTE "${file}")
      list(APPEND sources ${file})
    else()
      list(APPEND sources ${BASE}/${file})
    endif()
  endforeach()
  set(TARGET mecanum_f407_first_stage_4w20)
  add_executable(${TARGET} ${sources})
  foreach(property INCLUDE_DIRECTORIES COMPILE_DEFINITIONS COMPILE_OPTIONS)
    get_target_property(values mecanum_f407_four_link_groundb ${property})
    set_property(TARGET ${TARGET} PROPERTY ${property} "${values}")
  endforeach()
  get_target_property(options mecanum_f407_four_link_groundb LINK_OPTIONS)
  list(FILTER options EXCLUDE REGEX "^-Wl,-Map=")
  target_link_options(${TARGET} PRIVATE ${options} -Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/${TARGET}.map)
  set_target_properties(${TARGET} PROPERTIES SUFFIX ".elf")
  add_custom_command(TARGET ${TARGET} POST_BUILD
      COMMAND ${CMAKE_OBJCOPY} -O binary $<TARGET_FILE:${TARGET}> ${TARGET}.bin
      COMMAND ${CMAKE_SIZE} $<TARGET_FILE:${TARGET}>)
else()
  set(TARGET test_first_stage_4w20)
  add_executable(${TARGET} tests/test_first_stage.c ${ALGORITHMS})
  target_compile_options(${TARGET} PRIVATE -std=c11 -Wall -Wextra -Wpedantic -Werror -g
      -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(${TARGET} PRIVATE -fsanitize=address,undefined)
endif()
target_include_directories(${TARGET} BEFORE PRIVATE ${CMAKE_CURRENT_LIST_DIR}/Inc ${ACCEPTED}/Inc)
target_include_directories(${TARGET} PRIVATE ${BASE}/Inc ${WORKSPACE}/protocol)
target_compile_definitions(${TARGET} PRIVATE FOUR_LINK_MOTION=1 FOUR_LINK_GROUNDB=1
    DRIVE_BENCH=1 DRIVE_STARTUP_OBSERVE=1 FOUR_CONTINUOUS_CANDIDATE=1 FOUR_HIGHSPEED_CANDIDATE=1)
target_link_libraries(${TARGET} PRIVATE m)
