cmake_minimum_required(VERSION 3.20)
option(CONTINUOUS_ARM "Compile an OFFLINE firmware candidate, never flash" OFF)
option(CONTINUOUS_SANITIZERS "Native ASan/UBSan" ON)
get_filename_component(BASE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
get_filename_component(WORKSPACE "${BASE}/../.." ABSOLUTE)
file(READ ${CMAKE_CURRENT_LIST_DIR}/baseline.json BASELINE)
string(JSON BASELINE_LENGTH LENGTH "${BASELINE}")
math(EXPR BASELINE_LAST "${BASELINE_LENGTH}-1")
foreach(index RANGE ${BASELINE_LAST})
  string(JSON file MEMBER "${BASELINE}" ${index})
  string(JSON expected GET "${BASELINE}" "${file}")
  file(SHA256 ${BASE}/${file} actual)
  if(NOT actual STREQUAL expected)
    message(FATAL_ERROR "Legacy baseline drift: ${file}; review candidate overlay before building")
  endif()
endforeach()
if(CONTINUOUS_ARM)
  include(${BASE}/cmake/target.cmake)
  include(${BASE}/cmake/gcc-arm-none-eabi.cmake)
  set(CMAKE_C_COMPILER_FORCED TRUE)
  include(${CMAKE_CURRENT_LIST_DIR}/cmake/absolute_compiler.cmake)
endif()
project(continuous_4wgc_offline C ASM)
set(CANDIDATE_DEFS FOUR_LINK_MOTION=1 FOUR_LINK_GROUNDB=1 DRIVE_BENCH=1
    DRIVE_STARTUP_OBSERVE=1 FOUR_CONTINUOUS_CANDIDATE=1)
set(OVERLAY Src/four_link_motion.c Src/four_drive_group.c
    Src/four_wheel_speed.c Src/drive_bench_control.c)
if(CONTINUOUS_ARM)
  # Import target metadata only. The accepted image is neither built nor
  # edited; every build product stays under this NEW binary directory.
  set(CMAKE_PROJECT_INCLUDE_BEFORE ${CMAKE_CURRENT_LIST_DIR}/cmake/absolute_compiler.cmake)
  add_subdirectory(${BASE} ${CMAKE_CURRENT_BINARY_DIR}/legacy-metadata EXCLUDE_FROM_ALL)
  unset(CMAKE_PROJECT_INCLUDE_BEFORE)
  get_target_property(LEGACY_SOURCES mecanum_f407_four_link_groundb SOURCES)
  set(SOURCES)
  foreach(file IN LISTS LEGACY_SOURCES)
    if(file IN_LIST OVERLAY)
      list(APPEND SOURCES ${CMAKE_CURRENT_LIST_DIR}/${file})
    elseif(IS_ABSOLUTE "${file}")
      list(APPEND SOURCES ${file})
    else()
      list(APPEND SOURCES ${BASE}/${file})
    endif()
  endforeach()
  set(TARGET mecanum_f407_continuous_4wgc_OFFLINE)
  add_executable(${TARGET} ${SOURCES})
  foreach(property INCLUDE_DIRECTORIES COMPILE_DEFINITIONS COMPILE_OPTIONS)
    get_target_property(values mecanum_f407_four_link_groundb ${property})
    set_property(TARGET ${TARGET} PROPERTY ${property} "${values}")
  endforeach()
  get_target_property(options mecanum_f407_four_link_groundb LINK_OPTIONS)
  list(FILTER options EXCLUDE REGEX "^-Wl,-Map=")
  list(APPEND options -Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/${TARGET}.map)
  target_link_options(${TARGET} PRIVATE ${options})
  set_target_properties(${TARGET} PROPERTIES SUFFIX ".elf")
  add_custom_command(TARGET ${TARGET} POST_BUILD
      COMMAND ${CMAKE_OBJCOPY} -O binary $<TARGET_FILE:${TARGET}> ${TARGET}.bin
      COMMAND ${CMAKE_SIZE} $<TARGET_FILE:${TARGET}>
      COMMENT "OFFLINE 4WGC candidate only; NO FLASH COMMAND EXISTS")
else()
  set(TARGET test_continuous_4wgc)
  add_executable(${TARGET} tests/test_continuous.c ${OVERLAY}
      ${BASE}/Src/mecanum_drive_plan.c ${BASE}/Src/mecanum_kinematics.c
      ${BASE}/Src/mc_protocol.c)
  target_compile_options(${TARGET} PRIVATE -std=c11 -Wall -Wextra -Wpedantic -Werror -g)
  if(CONTINUOUS_SANITIZERS)
    target_compile_options(${TARGET} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${TARGET} PRIVATE -fsanitize=address,undefined)
  endif()
  enable_testing()
  add_test(NAME continuous_4wgc COMMAND ${TARGET})
endif()
target_include_directories(${TARGET} BEFORE PRIVATE ${CMAKE_CURRENT_LIST_DIR}/Inc)
target_include_directories(${TARGET} PRIVATE ${BASE}/Inc ${WORKSPACE}/protocol)
target_compile_definitions(${TARGET} PRIVATE ${CANDIDATE_DEFS})
target_link_libraries(${TARGET} PRIVATE m)
